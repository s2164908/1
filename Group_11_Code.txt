model Model
uses "mmxprs"

declarations
  !Variables from TomMac.dat
  Products:   set of string
  Varieties:  set of string
  Contents:   set of string
  ProductReduction:       array(Products) of real
  OtherIngredientFactor:  array(Products) of real
  OtherIngredientCost:    array(Products) of real
  ProductionCost:         array(Products) of real
  TomatoProperties:       array(Contents, Varieties) of real
  ProductProperties:      array(Contents, Products) of real
  StorageCost:            real
  WarehouseSize:          real
  NewWarehouseSize:       real
  NewWarehouseCost:       real
  NewProductionIncrease:  real
  NewProductionCost:      real

  !Add integer set to number quarters
  NQuarters = 4
  Quarters = 1..NQuarters

  !Variables from TomMacQuaters.dat
  PeriodNames:  set of string
  TomatoSupply: array(Varieties, Quarters) of real
  TomatoCost:   array(Varieties, Quarters) of real
  Demand:           array(Products) of real
  SalePrice:        array(Products) of real
  Contract:         array(Products) of real
  ContractPrice:    array(Products) of real
  NewContract:      array(Products) of real
  NewContractPrice: array(Products) of real
  ProductionLimit:  array(Products) of real

  !Define decision variables
  Use:    array(Varieties, Products, Quarters) of mpvar !kilos of tomato varieties used per product per quarter
  Make:   array(Products, Quarters) of mpvar            !kilos of products made per quarter
  Sell:   array(Products, Quarters) of mpvar            !kilos of products sold per quarter
  Stock:  array(Products, Quarters) of mpvar            !kilos of products in stock per quarter
end-declarations

initialisations from "TomMac.dat"
    Products Varieties Contents ProductReduction OtherIngredientFactor OtherIngredientCost ProductionCost TomatoProperties ProductProperties
    StorageCost WarehouseSize NewWarehouseSize NewWarehouseCost NewProductionIncrease NewProductionCost
end-initialisations

initialisations from "TomMacQuarters.dat"
  PeriodNames TomatoSupply TomatoCost Demand SalePrice Contract ContractPrice NewContract NewContractPrice ProductionLimit
end-initialisations

!declare procedures used in code
forward procedure feasibility               !used to check feasibility of LP
forward procedure product_data(pr: string)  !used for generating CSV data for products
forward procedure strategicdecision         !used for testing scenarios

!helper variable: 1 for adding non-negative quarterly profit, 0 for allowing non-negative quarterly profit
NonNegQProfit := 0

!Base case
!define all additional variables
forall(q in Quarters) do
  !product-related variables
  forall(pr in Products) do
    OtherIngredientCostQuarter(pr, q) := OtherIngredientCost(pr) *
                                          OtherIngredientFactor(pr) *
                                          ProductReduction(pr) *
                                          sum(v in Varieties) Use(v, pr, q) !total spent on other ingredients per product per quarter

    ProductionCostQuarter(pr, q) := ProductionCost(pr) * Make(pr,q) !total spent on production per product per quarter
    MakeCostQuarter(pr, q) := OtherIngredientCostQuarter(pr, q) + ProductionCostQuarter(pr, q)  !cost of product produced per quarter
	  StockCost(pr, q) := Stock(pr, q) * StorageCost  !calculate cost of quarterly stock of products
  end-do

  !tomato variety related variables
  forall(v in Varieties)
    TotalTomatoCostQuarter(v, q) := sum(pr in Products) Use(v,pr,q) * TomatoCost(v,q)  !total cost of tomato variety per quarter

  !calculate quarterly income, cost and profit
  Income(q) := sum(pr in Products) (Contract(pr)*ContractPrice(pr) +
                                      (Sell(pr,q)-Contract(pr))*SalePrice(pr)) !quarterly income
  Cost(q) := sum(v in Varieties) TotalTomatoCostQuarter(v, q) +
              sum(pr in Products) MakeCostQuarter(pr, q) +
              sum(pr in Products) StockCost(pr, q) !quarterly cost
  Profit(q) := Income(q) - Cost(q) !quarterly profit
end-do


!define the objective
TotalIncome := sum(q in Quarters) Income(q)
TotalCost := sum(q in Quarters) Cost(q)
TotalProfit := TotalIncome - TotalCost


!set up the constraints
forall(q in Quarters) do
  !tomato related constraints
  forall(v in Varieties)
    SupplyCS(v,q) := sum(pr in Products) Use(v,pr,q) <= TomatoSupply(v,q)

  !product related constraints
  forall(pr in Products) do
    ProductionCS(pr, q) := Make(pr,q) <= ProductionLimit(pr)
    DemandCS(pr, q) := Sell(pr,q) - Contract(pr) <= Demand(pr)
    ContractCS(pr, q) := Sell(pr, q) >= Contract(pr)
    !product produced per quarter: P=(1+I)RM
    MakeCS(pr, q) := Make(pr, q) = (1+OtherIngredientFactor(pr)) *
                                    ProductReduction(pr) *
                                    sum(v in Varieties) Use(v, pr, q)

    !content makeup constraints (solid/liquid)
    forall(c in Contents) do
      !define additional variable for shorthand: MakeContent(c, pr, q)
      MakeContent(c, pr, q) := ((1+OtherIngredientFactor(pr)) * ProductReduction(pr) *
                                    sum(v in Varieties) (TomatoProperties(c, v) * Use(v, pr, q)))

      !define upper and lower bound constraints
      ContentLBCS(c, pr, q) := MakeContent(c, pr, q) >= ProductProperties(c, pr) * Make(pr, q)
      ContentUBCS(c, pr, q) := MakeContent(c, pr, q) <= ProductProperties(c, pr) * Make(pr, q)
    end-do

    !total content constraint
    TotalContentCS(pr, q) := sum(c in Contents) MakeContent(c, pr, q) = Make(pr, q)

    !stock constraints
    if q = 1 then
		  StockCS(pr, q) := Stock(pr, q) = Stock(pr, NQuarters) + Make(pr, q) - Sell(pr, q)
	  else
		  StockCS(pr, q) := Stock(pr, q) = Stock(pr, q-1) + Make(pr, q) - Sell(pr, q)
	  end-if
  end-do

  !warehouse size constraint
  WarehouseCS(q) := sum(pr in Products) Stock(pr, q) <= WarehouseSize
end-do

!if NonNegQProfit=1, add extra constraints
if NonNegQProfit = 1 then
  !force quarterly profit to be non-negative
  forall(q in Quarters)
    ProfitCS(q) := Profit(q) >= 0
  
  !additional constraint: force Paste production to equal production capacity in Q3
  !(otherwise would be false due to non-negative profit)
  PasteCS := Make("Paste", 3) = ProductionLimit("Paste")
end-if

!maximise objective
maximise(TotalProfit)

end-if

!procedure definition: check feasibility of LP
procedure feasibility
  if getprobstat = XPRS_OPT then
    status := "Optimum found"
  elif getprobstat = XPRS_UNF then
    status := "Unfinished"
  elif getprobstat = XPRS_INF then
    status := "Infeasible"
  elif getprobstat = XPRS_UNB then
    status := "Unbounded"
  elif getprobstat = XPRS_OTH then
    status := "Failed"
  else
    status := "???"
  end-if
  writeln("Problem status: ", status)
end-procedure


!procedure definition: export all generated data about a single product to a CSV
procedure product_data(pr: string)
  write("Quarter")
  write(",Production(kg),Sold(kg),Stock(kg),Demand(kg),Income(£),Total cost(£),Profit(£),Contract(kg)")
  forall(v in Varieties)
    write(",", v, " used(kg)")
  writeln
  forall(q in Quarters) do
    write(PeriodNames(q))
    write(",",strfmt(getsol(Make(pr,q)),0,1))
    write(",",strfmt(getsol(Sell(pr,q)),0,1))
    write(",",strfmt(getsol(Stock(pr,q)),0,1))
    write(",",strfmt(getsol(Sell(pr,q)-Contract(pr)),0,1))
    write(",",strfmt(getsol((Contract(pr)*ContractPrice(pr) + (Sell(pr,q)-Contract(pr))*SalePrice(pr))),0,1))
    write(",",strfmt(getsol(MakeCostQuarter(pr, q)+ StockCost(pr, q)+
                            sum(v in Varieties) Use(v,pr,q) * TomatoCost(v,q)),0,1))
    write(",",strfmt(getsol((Contract(pr)*ContractPrice(pr) + (Sell(pr,q)-Contract(pr))*SalePrice(pr))-
                            (MakeCostQuarter(pr, q)+ StockCost(pr, q)+ (sum(v in Varieties) Use(v,pr,q) * TomatoCost(v,q)))),0,1))
    write(",",strfmt(getsol(Sell(pr,q)-Demand(pr)),0,1))
    forall(v in Varieties)
      write(",", strfmt(getsol(Use(v,pr,q)),0,1))
  writeln
  end-do
end-procedure


!procefure definition: scenarios on strategic decisions
procedure strategicdecision
  initialisations from "TomMacPlan.dat"
      NewWarehouseSize NewWarehouseCost NewContract NewContractPrice NewProductionIncrease NewProductionCost
      Demand SalePrice ContractPrice ProductionCost
  end-initialisations

  !add delta for change of product properties
  delta := 0.02
  
  !keep track of highest profit in scenarios
  HighestProfit := getsol(TotalProfit)
  
  !NewContractDecision, IncreaseWarehouse, IncreaseProduction are taking values 0 or 1 depending on whether we want to make respective strategic decisions
  forall(NewContractDecision in 0..1) do
    forall(IncreaseWarehouse in 0..1) do
      forall(IncreaseProduction in 0..1) do
        !recalculate additional variables
        forall(q in Quarters) do
          forall(pr in Products) do
            ProductionCostQuarter(pr, q) := ProductionCost(pr) * Make(pr,q) !total spent on production per product per quarter
            MakeCostQuarter(pr, q) := OtherIngredientCostQuarter(pr, q) + ProductionCostQuarter(pr, q)  !cost of product produced per quarter
          end-do

          !recalculate the new quarterly income, cost and profit
          Income(q) := sum(pr in Products) (Contract(pr)*ContractPrice(pr) + NewContractDecision * (NewContract(pr)*NewContractPrice(pr))+
                                          (Sell(pr,q)-Contract(pr)- NewContractDecision * NewContract(pr))*SalePrice(pr)) !new quarterly income
          Cost(q) := sum(v in Varieties) TotalTomatoCostQuarter(v, q) +
                      sum(pr in Products) MakeCostQuarter(pr, q) +
                      sum(pr in Products) StockCost(pr, q) !new quarterly cost without annual payments
          
          !add annual costs to final quarter
          if q = 4 then
            Cost(4) += NewWarehouseCost * IncreaseWarehouse + NewProductionCost * IncreaseProduction
          end-if
          
          Profit(q) := Income(q) - Cost(q) !new quarterly profit
        end-do
        
        
        !redefine the objective
        NewTotalIncome := sum(q in Quarters) Income(q)
        NewTotalCost := sum(q in Quarters) Cost(q)
        NewTotalProfit := NewTotalIncome - NewTotalCost
        
        
        !redefine affected constraints
        forall(q in Quarters) do
          !product related constraints
          forall(pr in Products) do
            !update production limit
            ProductionCS(pr, q) := Make(pr, q) <= ProductionLimit(pr) * (1 + NewProductionIncrease / 100 * IncreaseProduction)
            DemandCS(pr, q) := Sell(pr,q) - Contract(pr) - NewContractDecision * NewContract(pr) <= Demand(pr)
            ContractCS(pr, q) := Sell(pr, q) >= Contract(pr) + NewContractDecision * NewContract(pr)

            !content makeup constraints (solid/liquid)
            forall(c in Contents) do
              !define upper and lower bound constraints
              ContentLBCS(c, pr, q) := MakeContent(c, pr, q) >= (ProductProperties(c, pr)-delta) * Make(pr, q)
              ContentUBCS(c, pr, q) := MakeContent(c, pr, q) <= (ProductProperties(c, pr)+delta) * Make(pr, q)
            end-do
          end-do
          
          !warehouse constraint
          WarehouseCS(q) := sum(pr in Products) Stock(pr, q) <= WarehouseSize + NewWarehouseSize * IncreaseWarehouse
        end-do
        
        
        !Maximise the new profit
        maximise(NewTotalProfit)
        
        
        !print the strategy
        writeln("Strategy:")

        if(IncreaseProduction = 1) then
          writeln("- Increasing production limit, ")
        else
          writeln("- Not increasing production limit, ")
        end-if

        if(IncreaseWarehouse = 1) then
          writeln("- Increasing warehouse, ")
        else
          writeln("- Not increasing warehouse, ")
        end-if

        if(NewContractDecision = 1) then
          writeln("- Accepting contract.")
        else
          writeln("- Not accepting contract.")
        end-if
        
        !Test if optimal solution is found
        feasibility
        
        !if optimal, print the result
        if status="Optimum found" then
          writeln("Total profit: £", strfmt(getsol(NewTotalProfit), 0, 1))
          
          !if new profit is bigger, update highest strategy profit
          if(getsol(NewTotalProfit) > getsol(HighestProfit) ) then
            HighestProfit := getsol(NewTotalProfit)
          end-if
        end-if
        
        writeln
      end-do
    end-do
  end-do
  
  !write highest strategy profit
  writeln("The highest profit achieved is £", strfmt(getsol(HighestProfit), 0, 1))
end-procedure
